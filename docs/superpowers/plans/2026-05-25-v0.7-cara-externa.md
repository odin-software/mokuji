## Task D1: Migration `0012_digest_frequency_and_onboarding.sql`

**Files:**

- [ ] **Step 1: Crear migration**

```sql
-- Migration 0012: digest_frequency opt-in (weekly default + daily/off) + onboarded_at gate.
--
-- Reemplaza la semántica de email_digest_enabled (boolean) por digest_frequency:
--   - 'weekly' (default, comportamiento actual)
--   - 'daily'  (opt-in del onboarding o settings)
--   - 'off'    (sin emails)
--
-- email_digest_enabled queda como columna por compatibilidad (no la borramos en esta
-- migration por seguridad; el worker la ignora a partir de 0012). Limpieza opcional
-- en migration futura cuando el codebase no la referencie más.
--
-- onboarded_at es el gate del wizard de v0.7. Users existentes son backfilled
-- con created_at para que NO se vean forzados a pasar por onboarding (predatory).

ALTER TABLE users ADD COLUMN digest_frequency TEXT NOT NULL DEFAULT 'weekly';
ALTER TABLE users ADD COLUMN onboarded_at INTEGER;

-- Backfill: traducir email_digest_enabled actual a la nueva columna.
-- - Si email_digest_enabled = 1 → digest_frequency = 'weekly' (ya está por DEFAULT).
-- - Si email_digest_enabled = 0 → digest_frequency = 'off'.
UPDATE users SET digest_frequency = 'off' WHERE email_digest_enabled = 0;

-- Backfill: users existentes no pasan por onboarding (sería predatorio).
UPDATE users SET onboarded_at = created_at WHERE onboarded_at IS NULL;

-- Índice para que el query del worker filtrando por frequency sea barato cuando crezcan los users.
CREATE INDEX IF NOT EXISTS idx_users_digest_frequency ON users(digest_frequency);
```

La nueva lógica: el worker tickea cada hora (ya lo hace). En cada tick, dispatch SOLO si:
- Estamos en 09-22 UTC (banda horaria sigue igual)
- Hay al menos un user con frequency='daily' Y last_digest < startOfDay, OR (es domingo Y hay user con frequency='weekly' Y last_digest < startOfWeek)

Simplificación: dejar la decisión SQL-side. Tickear siempre dentro de la banda 09-22 UTC, y que la query filtre quién recibe.

Reemplazar `tickIfDue` (línea 126 aprox) por:

```go
// tickIfDue evalúa si despachar en este tick. Banda horaria 09-22 UTC.
// La decisión de qué users recibir está en selectCandidates (SQL).
func (w *Worker) tickIfDue(ctx context.Context, now time.Time) {
    if now.Hour() < w.fireHourStart || now.Hour() > w.fireHourEnd {
        return
    }
    w.runDispatch(ctx, now)
}
```

```go
//   2. Qué monitoreamos (informativo — MangaDex + Asura).

package onboarding

    "errors"
    "fmt"
    "html/template"

    "log/slog"

    "time"

    "git.odin.do/kengru/mokuji/internal/version"
)

type Config struct {
    DB            *sql.DB
    HTMLTemplates fs.FS
    Logger        *slog.Logger
}

type Handler struct {
    db      *sql.DB
    tmpl    *template.Template
    logger  *slog.Logger
}

func New(cfg Config) (*Handler, error) {
    if cfg.DB == nil {
        return nil, errors.New("DB requerido")
    }
    if cfg.Logger == nil {
        cfg.Logger = slog.Default()
    }
    funcs := template.FuncMap{
        "csrfInput":  csrf.Input,
        "appVersion": func() string { return version.Value },
    }
    tmpl, err := template.New("base").Funcs(funcs).ParseFS(
        cfg.HTMLTemplates,
        "layout.html",
        "onboarding/step1.html",
        "onboarding/step2.html",
        "onboarding/step3.html",
    )
    // BLOQUEO: layout.html define los blocks "title"/"content". onboarding/*.html
    // los redefine. Igual que en home/, parsear cada step como un *template.Template
    // separado para evitar block collision.
    if err != nil {
        return nil, fmt.Errorf("parsear templates: %w", err)
    }
    return &Handler{db: cfg.DB, tmpl: tmpl, logger: cfg.Logger}, nil
}

func (h *Handler) Mount(mux *http.ServeMux) {
    mux.HandleFunc("GET /onboarding", h.step1)
    mux.HandleFunc("GET /onboarding/step2", h.step2)
    mux.HandleFunc("GET /onboarding/step3", h.step3)
    mux.HandleFunc("POST /onboarding/finish", h.finish)
}

type stepData struct {
    Session *session.Session
}

func (h *Handler) requireUnonboarded(w http.ResponseWriter, r *http.Request) (*session.Session, bool) {
    s, ok := session.FromContext(r.Context())
    if !ok {
        http.Redirect(w, r, "/auth", http.StatusSeeOther)
        return nil, false
    }
    var onboardedAt sql.NullInt64
    err := h.db.QueryRowContext(r.Context(),
        `SELECT onboarded_at FROM users WHERE id = ?`, s.UserID,
    ).Scan(&onboardedAt)
    if err != nil {
        h.logger.Error("cargar onboarded_at", "user_id", s.UserID, "err", err)
        http.Error(w, "internal error", http.StatusInternalServerError)
        return nil, false
    }
    if onboardedAt.Valid {
        http.Redirect(w, r, "/library", http.StatusSeeOther)
        return nil, false
    }
    return s, true
}

func (h *Handler) step1(w http.ResponseWriter, r *http.Request) {
    s, ok := h.requireUnonboarded(w, r)
    if !ok { return }
    h.render(w, "onboarding/step1.html", stepData{Session: s})
}

func (h *Handler) step2(w http.ResponseWriter, r *http.Request) {
    s, ok := h.requireUnonboarded(w, r)
    if !ok { return }
    h.render(w, "onboarding/step2.html", stepData{Session: s})
}

func (h *Handler) step3(w http.ResponseWriter, r *http.Request) {
    s, ok := h.requireUnonboarded(w, r)
    if !ok { return }
    h.render(w, "onboarding/step3.html", stepData{Session: s})
}

func (h *Handler) finish(w http.ResponseWriter, r *http.Request) {
    s, ok := h.requireUnonboarded(w, r)
    if !ok { return }
    freq := r.FormValue("digest_frequency")
    if freq != "weekly" && freq != "daily" && freq != "off" {
        freq = "weekly"
    }
    _, err := h.db.ExecContext(r.Context(),
        `UPDATE users SET digest_frequency = ?, onboarded_at = ? WHERE id = ?`,
        freq, time.Now().Unix(), s.UserID,
    )
    if err != nil {
        h.logger.Error("finish onboarding", "user_id", s.UserID, "err", err)
        http.Error(w, "internal error", http.StatusInternalServerError)
        return
    }
    http.Redirect(w, r, "/library", http.StatusSeeOther)
}

func (h *Handler) render(w http.ResponseWriter, name string, data interface{}) {
    w.Header().Set("Content-Type", "text/html; charset=utf-8")
    if err := h.tmpl.ExecuteTemplate(w, "layout", data); err != nil {
        h.logger.Error("renderizar onboarding", "template", name, "err", err)
    }
}
```

```go
// y los redirige a /onboarding, EXCEPTO rutas excluidas (el wizard mismo, /auth/logout,
// estáticos, etc.).
//
// Esto no necesita worker — la decisión es por-request via la sesión.
func (s *Server) onboardingRedirect(next http.Handler) http.Handler {
    excluded := map[string]struct{}{
        "/onboarding":           {},
        "/onboarding/step2":     {},
        "/onboarding/step3":     {},
        "/onboarding/finish":    {},
        "/auth":                 {},
        "/auth/start":           {},
        "/auth/verify":          {},
        "/auth/logout":          {},
        "/health":               {},
        "/version":              {},
        "/manifest.webmanifest": {},
        "/sw.js":                {},
    }
    return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
        // /static/, /u/ y rutas que no necesitamos gating
        if strings.HasPrefix(r.URL.Path, "/static/") || strings.HasPrefix(r.URL.Path, "/u/") || strings.HasPrefix(r.URL.Path, "/unsubscribe") {
            next.ServeHTTP(w, r)
            return
        }
        if _, ok := excluded[r.URL.Path]; ok {
            next.ServeHTTP(w, r)
            return
        }
        sess, ok := session.FromContext(r.Context())
        if !ok {
            // Sin sesión → ya no nos toca, no redirigimos.
            next.ServeHTTP(w, r)
            return
        }
        var onboardedAt sql.NullInt64
        err := s.cfg.DB.QueryRowContext(r.Context(),
            `SELECT onboarded_at FROM users WHERE id = ?`, sess.UserID,
        ).Scan(&onboardedAt)
        if err != nil {
            // Mejor pass-through que romper el flujo si la query falla.
            next.ServeHTTP(w, r)
            return
        }
        if !onboardedAt.Valid {
            http.Redirect(w, r, "/onboarding", http.StatusSeeOther)
            return
        }
        next.ServeHTTP(w, r)
    })
}
```

NOTA: este middleware requiere que `server.Server.cfg` exponga `DB *sql.DB`. Verificar — si no, agregar el field a `Config`.
